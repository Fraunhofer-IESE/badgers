    sampling

    Sampling strategies for generating values within or outside observed distributions.

    WithinDistributionSampler subclasses generate plausible values from the observed data distribution (used for exogenous nodes in causal generation, and for within-distribution value synthesis: InstanceSampler copies real values per cell).

    OutOfDistributionSampler subclasses generate anomalous values outside the observed distribution (used for perturbation nodes in causal generation).

    class HypersphereSampler(OutOfDistributionSampler):
    """Sample points on a hypersphere with radius >= 3 sigma.

    Points are generated on a hypersphere with radius
    ``3 + exponential(scale)`` in standardized space, then transformed
    back to the original scale.

    Parameters
    ----------
    scale : float, default=1.0
        Scale parameter for the exponential distribution.
    """

    def __init__(self, scale=1.0):
        self.scale = scale

    def sample(self, rng, X, columns, n_samples):
        n_features = len(columns)
        means = X[:, columns].mean(axis=0)
        stds = X[:, columns].std(axis=0)
        stds[stds == 0] = 1.0
        radii = 3.0 + rng.exponential(scale=self.scale, size=n_samples)
        standardized = random_spherical_coordinates(
            random_generator=rng,
            size=n_features,
            radii=radii,
        )
        return means + standardized * stds

    class IndependentHistogramSampler(OutOfDistributionSampler):
    """Sample each requested column from its low-density histogram bins.

    Parameters
    ----------
    bins : int, default=10
        Number of histogram bins per column. Must be positive.
    """

    def __init__(self, bins=10):
        if bins <= 0:
            raise ValueError(f"bins must be > 0, got {bins}")
        self.bins = bins

    def sample(self, rng, X, columns, n_samples):
        result = np.empty((n_samples, len(columns)), dtype=float)
        for output_column, input_column in enumerate(columns):
            counts, edges = np.histogram(X[:, input_column], bins=self.bins)
            probabilities = 1.0 - counts / counts.max()
            probability_sum = probabilities.sum()
            if probability_sum == 0:
                probabilities = np.full(self.bins, 1.0 / self.bins)
            else:
                probabilities /= probability_sum
            bin_indices = rng.choice(
                self.bins, size=n_samples, p=probabilities
            )
            result[:, output_column] = rng.uniform(
                low=edges[bin_indices], high=edges[bin_indices + 1]
            )
        return result

    Per-cell bootstrap: for every (sample, column) pair an independent row is drawn from the full data and that cell's value is copied. Every returned value is a real value of its column, but the cells of one row come from different rows, so each row is a plausible-looking recombination — within-distribution per cell, unlikely as a row.

    class InstanceSampler(WithinDistributionSampler):
    """Sample each cell by copying the value of a random real row.

    Per-cell bootstrap: for every (sample, column) pair an independent
    row is drawn from the full data and that cell's value is copied.
    Every returned value is a real value of its column, but the cells of
    one row come from different rows, so each row is a plausible-looking
    recombination — within-distribution per cell, unlikely as a row.
    """

    def sample(self, rng, X, columns, n_samples):
        row_indices = rng.integers(0, len(X), size=(n_samples, len(columns)))
        return X[row_indices, np.asarray(columns)]

    class JointHistogramSampler(OutOfDistributionSampler):
    """Sample requested columns jointly from low-density histogram bins.

    Parameters
    ----------
    bins : int, default=10
        Number of bins along each standardized dimension. Must be positive.
    threshold_low_density : float, default=0.1
        Maximum normalized bin count considered low density. Must be
        strictly between zero and one.
    """

    def __init__(self, bins=10, threshold_low_density=0.1):
        if bins <= 0:
            raise ValueError(f"bins must be > 0, got {bins}")
        if not 0 < threshold_low_density < 1:
            raise ValueError(
                "threshold_low_density must be between 0 and 1, "
                f"got {threshold_low_density}"
            )
        self.bins = bins
        self.threshold_low_density = threshold_low_density

    def sample(self, rng, X, columns, n_samples):
        if len(columns) > 5:
            raise NotImplementedError(
                "Joint histogram sampling supports at most 5 columns"
            )

        target = X[:, columns]
        scaler = StandardScaler()
        standardized = scaler.fit_transform(target)
        counts, edges = np.histogramdd(
            standardized, bins=self.bins
        )
        normalized_counts = counts / counts.max()
        low_density_bins = np.argwhere(
            normalized_counts <= self.threshold_low_density
        )
        if len(low_density_bins) == 0:
            raise ValueError(
                "threshold_low_density found no low-density region"
            )

        selected_bins = low_density_bins[
            rng.integers(0, len(low_density_bins), size=n_samples)
        ]
        samples = np.column_stack([
            rng.uniform(
                low=edges[column][selected_bins[:, column]],
                high=edges[column][selected_bins[:, column] + 1],
            )
            for column in range(len(columns))
        ])
        return scaler.inverse_transform(samples)

    KernelDensitySampler

    Bases: OutOfDistributionSampler

    Sample from low-density regions of a kernel density estimate.

    The requested columns are standardized, a KernelDensity is fit on them (ball-tree accelerated), and the standardized bounding box padded by one standard deviation per side is discretized into a grid of at most grid_budget cells. Every cell is scored once, and the requested samples are drawn from a categorical over the cells at or below the threshold_low_density percentile of the data's log-densities — each point independently uniform inside its cell. There is no rejection step, so exactly n_samples points are always returned.

    The grid keeps at most grid_budget cells per dimension root (g = ceil(grid_budget ** (1 / d))) and at most 65536 cells in total, so the sampler is intended for low dimensionality (d <= ~16 with the default budget).

    Parameters

    threshold_low_density : float, default=0.1 Percentile (in [0, 1)) of the data's log-densities that defines the low-density boundary. grid_budget : int, default=2048 Approximate number of grid cells (g ** d). Must be >= 1.

    class KernelDensitySampler(OutOfDistributionSampler):
    """Sample from low-density regions of a kernel density estimate.

    The requested columns are standardized, a ``KernelDensity`` is fit on
    them (ball-tree accelerated), and the standardized bounding box padded
    by one standard deviation per side is discretized into a grid of at
    most ``grid_budget`` cells. Every cell is scored once, and the
    requested samples are drawn from a categorical over the cells at or
    below the ``threshold_low_density`` percentile of the data's
    log-densities — each point independently uniform inside its cell. There is no
    rejection step, so exactly ``n_samples`` points are always returned.

    The grid keeps at most ``grid_budget`` cells per dimension root
    (``g = ceil(grid_budget ** (1 / d))``) and at most 65536 cells in
    total, so the sampler is intended for low dimensionality (d <= ~16
    with the default budget).

    Parameters
    ----------
    threshold_low_density : float, default=0.1
        Percentile (in [0, 1)) of the data's log-densities that defines
        the low-density boundary.
    grid_budget : int, default=2048
        Approximate number of grid cells (``g ** d``). Must be >= 1.
    """

    def __init__(self, threshold_low_density=0.1, grid_budget=2048):
        if not 0 <= threshold_low_density < 1:
            raise ValueError(
                "threshold_low_density must be in [0, 1), "
                f"got {threshold_low_density}")
        if grid_budget < 1:
            raise ValueError(f"grid_budget must be >= 1, got {grid_budget}")
        self.threshold_low_density = threshold_low_density
        self.grid_budget = grid_budget

    def sample(self, rng, X, columns, n_samples):
        target = X[:, columns]
        d = target.shape[1]
        if n_samples == 0:
            return np.empty((0, d))
        scaler = StandardScaler()
        standardized = scaler.fit_transform(target)

        estimator = KernelDensity(
            bandwidth="scott", algorithm="ball_tree").fit(standardized)
        low_density_threshold = np.percentile(
            estimator.score_samples(standardized), self.threshold_low_density)

        g = min(int(np.ceil(self.grid_budget ** (1.0 / d))), 32)
        while g > 1 and g ** d > _MAX_GRID_CELLS:
            g -= 1
        if g < 2:
            raise NotImplementedError(
                f"a grid with >= 2 cells per dimension is infeasible for "
                f"{d} dimensions with grid_budget={self.grid_budget}; "
                f"use fewer columns or a larger budget")
        edges = [
            np.linspace(standardized[:, j].min() - 1.0,
                        standardized[:, j].max() + 1.0, g + 1)
            for j in range(d)
        ]
        total = g ** d
        idx = np.unravel_index(np.arange(total), (g,) * d)
        cell_lo = np.column_stack([edges[j][idx[j]] for j in range(d)])
        cell_hi = np.column_stack([edges[j][idx[j] + 1] for j in range(d)])
        centers = (cell_lo + cell_hi) / 2

        f_grid = estimator.score_samples(centers)
        weights = (f_grid <= low_density_threshold).astype(float)
        if weights.sum() == 0:
            # Pathological (tiny grid / very peaked data): fall back to
            # the least dense cells.
            weights[np.argsort(f_grid)[: min(len(f_grid), self.grid_budget)]] = 1.0
        cells = rng.choice(total, size=n_samples, p=weights / weights.sum())
        samples = np.column_stack([
            rng.uniform(cell_lo[cells, j], cell_hi[cells, j]) for j in range(d)
        ])
        return scaler.inverse_transform(samples)

    TreeCellSampler

    Bases: OutOfDistributionSampler

    Sample from low-density cells of an adaptive kd partition.

    The requested columns are standardized and partitioned by recursive median splits into cells holding at most leaf_size points each: the resolution adapts (fine in dense regions, coarse in sparse ones), so the cell density count / volume is a data-driven histogram with the bin size chosen by the data. Cells are extended axis-aligned to the midplanes with their neighbouring points (an axis-aligned Voronoi approximation); at the padded box boundary, where there is no neighbour to share with, a cell takes the whole remaining gap. Space that is still not covered by any extended cell (typically the box corners far from all points) is added back as empty count-0 cells: it is genuinely low density and always qualifies. A leaf cell is low density when the density of its extended territory (count over extended volume) is at most threshold_low_density times the global mean density, so a sparse fringe cell can qualify together with the empty space it borders; the requested samples are drawn from a categorical over all qualifying cells, weighted by volume (uniform over the qualifying space, like a fine regular grid), each point independently uniform inside its cell, so exactly n_samples points are always returned.

    Parameters

    threshold_low_density : float, default=0.1 Maximum cell density, as a share of the global mean density, for the cell to count as low density (in (0, 1]). leaf_size : int, default=8 Maximum number of points per leaf cell (must be >= 1).

    class TreeCellSampler(OutOfDistributionSampler):
    """Sample from low-density cells of an adaptive kd partition.

    The requested columns are standardized and partitioned by recursive
    median splits into cells holding at most ``leaf_size`` points each:
    the resolution adapts (fine in dense regions, coarse in sparse ones),
    so the cell density ``count / volume`` is a data-driven histogram
    with the bin size chosen by the data. Cells are extended axis-aligned
    to the midplanes with their neighbouring points (an axis-aligned
    Voronoi approximation); at the padded box boundary, where there is no
    neighbour to share with, a cell takes the whole remaining gap. Space
    that is still not covered by any extended cell (typically the box
    corners far from all points) is added back as empty count-0 cells:
    it is genuinely low density and always qualifies. A leaf cell is low
    density when the density of its extended territory (count over
    extended volume) is at most ``threshold_low_density`` times the
    global mean density, so a sparse fringe cell can qualify together
    with the empty space it borders; the requested samples are drawn
    from a categorical over all qualifying cells, weighted by volume
    (uniform over the qualifying space, like a fine regular grid), each
    point independently uniform inside its cell, so exactly
    ``n_samples`` points are always returned.

    Parameters
    ----------
    threshold_low_density : float, default=0.1
        Maximum cell density, as a share of the global mean density,
        for the cell to count as low density (in (0, 1]).
    leaf_size : int, default=8
        Maximum number of points per leaf cell (must be >= 1).
    """

    def __init__(self, threshold_low_density=0.1, leaf_size=8):
        if not 0 < threshold_low_density <= 1:
            raise ValueError(
                "threshold_low_density must be in (0, 1], "
                f"got {threshold_low_density}")
        if leaf_size < 1:
            raise ValueError(f"leaf_size must be >= 1, got {leaf_size}")
        self.threshold_low_density = threshold_low_density
        self.leaf_size = leaf_size

    def sample(self, rng, X, columns, n_samples):
        target = X[:, columns]
        d = target.shape[1]
        if n_samples == 0:
            return np.empty((0, d))
        scaler = StandardScaler()
        standardized = scaler.fit_transform(target)
        n = standardized.shape[0]

        lo0 = standardized.min(axis=0)
        hi0 = standardized.max(axis=0)
        leaves = list(_kd_partition(standardized, lo0, hi0, self.leaf_size))
        counts = np.array([c for c, _, _ in leaves], dtype=float)
        cell_lo = np.array([lo for _, lo, _ in leaves])
        cell_hi = np.array([hi for _, _, hi in leaves])

        # Density proxy: points per cell volume, relative to the global
        # mean density (degenerate zero-volume cells are infinitely dense)
        volumes = np.prod(cell_hi - cell_lo, axis=1)
        mean_volume = float(np.prod(hi0 - lo0))
        mean_density = n / mean_volume if mean_volume > 0 else 0.0
        density = np.full(len(volumes), np.inf)
        np.divide(counts, volumes, out=density, where=volumes > 0)

        # Extend each cell axis-aligned to the midplane with the
        # neighbouring points (empty space is shared with neighbours).
        # At the box boundary there is no neighbour to share with, so a
        # cell takes the whole remaining gap up to the padded edge.
        for j in range(d):
            values = np.sort(standardized[:, j])
            # side="right": nearest data point STRICTLY beyond the cell
            # edge (the edge itself is a data value for inner cells), so
            # both sides grow symmetrically; at the box boundary there is
            # no such point and the cell takes the full remaining gap.
            k = np.searchsorted(values, cell_hi[:, j], side="right")
            gap_plus = np.where(
                k < n, values[np.minimum(k, n - 1)] - cell_hi[:, j],
                (hi0[j] + 1.0) - cell_hi[:, j])
            extend_plus = np.where(k < n, 0.5 * gap_plus, gap_plus)
            k2 = np.searchsorted(values, cell_lo[:, j], side="left")
            gap_minus = np.where(
                k2 > 0, cell_lo[:, j] - values[k2 - 1],
                cell_lo[:, j] - (lo0[j] - 1.0))
            extend_minus = np.where(k2 > 0, 0.5 * gap_minus, gap_minus)
            cell_hi[:, j] = np.minimum(cell_hi[:, j] + extend_plus, hi0[j] + 1.0)
            cell_lo[:, j] = np.maximum(cell_lo[:, j] - extend_minus, lo0[j] - 1.0)

        # Space not covered by any extended cell (typically box corners
        # far from all points) cannot be reached by the sampler; detect
        # it on a coarse grid and add it back as empty count-0 cells
        box_lo = lo0 - 1.0
        box_hi = hi0 + 1.0
        g = min(int(np.ceil(_EMPTY_CELL_GRID_BUDGET ** (1 / d))), 64)
        edges = [np.linspace(box_lo[j], box_hi[j], g + 1) for j in range(d)]
        mids = [(e[:-1] + e[1:]) / 2 for e in edges]
        mesh = np.meshgrid(*mids, indexing="ij")
        grid_centers = np.stack([m.ravel() for m in mesh], axis=1)
        covered = np.zeros(len(grid_centers), dtype=bool)
        block = max(1, 2 ** 22 // max(len(grid_centers) * d, 1))
        for start in range(0, len(cell_lo), block):
            lo_b = cell_lo[start:start + block, None, :]
            hi_b = cell_hi[start:start + block, None, :]
            covered |= ((grid_centers[None] >= lo_b)
                        & (grid_centers[None] <= hi_b)).all(axis=2).any(axis=0)
        if not covered.all():
            uidx = np.flatnonzero(~covered)
            gidx = np.unravel_index(uidx, (g,) * d)
            empty_lo = np.stack([edges[j][gidx[j]] for j in range(d)], axis=1)
            empty_hi = np.stack([edges[j][gidx[j] + 1] for j in range(d)], axis=1)
            cell_lo = np.vstack([cell_lo, empty_lo])
            cell_hi = np.vstack([cell_hi, empty_hi])

        # Qualify leaves on the density of their extended territory
        # (count over extended volume): the extended area around a sparse
        # fringe cell belongs to it, so it qualifies together with the
        # empty space it borders. Empty cells always qualify. Weight all
        # qualifying cells by (extended) volume: the sampler is then
        # uniform over the qualifying space, like a fine regular grid
        leaf_volumes = np.prod(cell_hi[:len(density)] - cell_lo[:len(density)], axis=1)
        extended_density = np.full(len(density), np.inf)
        np.divide(counts, leaf_volumes, out=extended_density, where=leaf_volumes > 0)
        all_volumes = np.prod(cell_hi - cell_lo, axis=1)
        weights = np.where(
            extended_density <= self.threshold_low_density * mean_density,
            all_volumes[:len(density)], 0.0)
        weights = np.concatenate([weights, all_volumes[len(density):]])
        if weights.sum() == 0:
            # Pathological (e.g. very small data): fall back to the least
            # dense cells.
            fallback = np.argsort(extended_density)[: min(len(density), 10)]
            weights[fallback] = all_volumes[fallback]
            weights[weights == 0] = 1.0
        cells = rng.choice(len(weights), size=n_samples, p=weights / weights.sum())
        samples = np.column_stack([
            rng.uniform(cell_lo[cells, j], cell_hi[cells, j]) for j in range(d)
        ])
        return scaler.inverse_transform(samples)

    class UniformOutOfDistributionSampler(OutOfDistributionSampler):
    """Sample uniformly around the observed range.

    Samples from [min - expansion*range, max + expansion*range] for each
    column independently.

    Parameters
    ----------
    expansion : float
        Fraction to expand beyond [min, max]. Must be >= 0.
        ``0`` samples the exact data bounding box [min, max]
        (in-distribution by construction; the out-of-distribution-ness is
        the caller's choice).
    """

    def __init__(self, expansion):
        if expansion < 0:
            raise ValueError(
                f"expansion must be >= 0 for uniform sampling, got {expansion}"
            )
        self.expansion = expansion

    def sample(self, rng, X, columns, n_samples):
        mins = X[:, columns].min(axis=0)
        maxs = X[:, columns].max(axis=0)
        ranges = maxs - mins
        low = mins - self.expansion * ranges
        high = maxs + self.expansion * ranges
        return rng.uniform(low=low, high=high, size=(n_samples, len(columns)))

    class ZScoreSampler(OutOfDistributionSampler):
    """Sample using z-score strategy: ±(3 + exponential) * sigma per column.

    Each column independently gets a random sign, then a value of
    ``3 + exponential(scale)`` standard deviations away from the mean.

    Parameters
    ----------
    scale : float, default=1.0
        Scale parameter for the exponential distribution.
    """

    def __init__(self, scale=1.0):
        self.scale = scale

    def sample(self, rng, X, columns, n_samples):
        n_features = len(columns)
        means = X[:, columns].mean(axis=0)
        stds = X[:, columns].std(axis=0)
        stds[stds == 0] = 1.0
        signs = random_sign(rng, size=(n_samples, n_features))
        exponentials = rng.exponential(
            scale=self.scale, size=(n_samples, n_features)
        )
        standardized = signs * (3.0 + exponentials)
        return means + standardized * stds
